@wait-world
@sleep 8
speed 0
@set MALZIN where Malzin ~ Malzin (#\d+)
spawn "Hungry Bandit" Drifters near Malzin dist 15 count 1 ~ spawned 1/1
@set K chars 60 ~ Hungry Bandit (#\d+)

@sleep 20
speed 0
@set KNAME where ${K} ~ ^(.+?) #\d+

@sleep 4
speed 0
# run m10: the old regex captured " Prisoner Cage dist"; take the name between "| " and " dist="
@set CAGE buildings 400 Cage ~ \| ([A-Za-z ]*Cage[A-Za-z ]*?) dist=
order Malzin LIFT_PERSON target ${K}
speed 2
@sleep 10
order Malzin PUT_IN_CAGE building "${CAGE}"
@sleep 60
speed 0
where Malzin
where ${K}
screenshot rel-p4-03 ~ png
@log D:\Steam\steamapps\common\Kenshi\RE_Kenshi\mods\Stobe\stobe.log ~ SOCIAL_CAPTURE: structured kind=placed seq=\d+ load=\d+ actor=${MALZIN} target=${K} witnesses=d+ facts="place":"prison"

@until 300 where ${K} ~ dist=[0-9.]+$

@log D:\Steam\steamapps\common\Kenshi\RE_Kenshi\mods\Stobe\stobe.log ~ SOCIAL_CAPTURE: structured kind=recovered seq=\d+ load=\d+ actor=#0 target=${K}

@sleep 6
